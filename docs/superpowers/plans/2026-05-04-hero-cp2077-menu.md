# Hero CP2077 Menu Overlay — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Remplacer le contenu centré du hero par un menu d'accueil Cyberpunk 2077 avec transitions scrub vidéo + scroll programmatique.

**Architecture:** Le menu est un overlay React positionné différemment sur desktop (latéral gauche) et mobile (bottom). Un hook `useHeroTransition` orchestre le verrouillage des inputs, le scrub GSAP de la vidéo, et le scroll Lenis. Le `HeroBackground` expose une méthode `scrubToProgress` via ref forwarding/imperative handle.

**Tech Stack:** Next.js 15, React 19, TypeScript, TailwindCSS v4, GSAP, Lenis, Framer Motion.

### Task 6: Créer le hook useHeroTransition

**Files:**
- Create: `app/hooks/useHeroTransition.ts`

- [ ] **Step 1: Implémenter useHeroTransition**

```typescript
"use client";

import { useState, useCallback, useRef } from "react";
import { gsap } from "gsap";

interface UseHeroTransitionOptions {
  videoRef: React.RefObject<HTMLVideoElement | null>;
  lenisRef?: React.RefObject<{ scrollTo: (target: string | number | HTMLElement, options?: object) => void } | null>;
  videoDuration: number;
  isReducedMotion: boolean;
}

export function useHeroTransition({
  videoRef,
  lenisRef,
  videoDuration,
  isReducedMotion,
}: UseHeroTransitionOptions) {
  const [isTransitioning, setIsTransitioning] = useState(false);
  const transitionRef = useRef<gsap.core.Tween | null>(null);

  const navigateTo = useCallback(
    (targetProgress: number, sectionId?: string) => {
      if (isTransitioning || videoDuration === 0) return;

      const video = videoRef.current;
      if (!video) return;

      setIsTransitioning(true);

      const targetTime = Math.max(0, Math.min(videoDuration, targetProgress * videoDuration));
      const currentTime = video.currentTime;

      if (isReducedMotion) {
        // Instantané en reduced motion
        video.currentTime = targetTime;
        if (sectionId) {
          const el = document.getElementById(sectionId);
          if (el) el.scrollIntoView({ behavior: "auto" });
        }
        setIsTransitioning(false);
        return;
      }

      // Scrub animé avec GSAP
      const scrubObj = { t: currentTime };
      transitionRef.current = gsap.to(scrubObj, {
        t: targetTime,
        duration: 0.6,
        ease: "power2.inOut",
        onUpdate: () => {
          video.currentTime = scrubObj.t;
        },
        onComplete: () => {
          if (sectionId) {
            const el = document.getElementById(sectionId);
            if (el) {
              // Utiliser Lenis si disponible, sinon scroll natif
              if (lenisRef?.current) {
                lenisRef.current.scrollTo(el, { offset: 0 });
              } else {
                el.scrollIntoView({ behavior: "smooth" });
              }
            }
          }
          // Débloquer après un délai pour laisser le scroll démarrer
          setTimeout(() => {
            setIsTransitioning(false);
          }, 400);
        },
      });
    },
    [isTransitioning, videoDuration, videoRef, lenisRef, isReducedMotion]
  );

  const cancelTransition = useCallback(() => {
    if (transitionRef.current) {
      transitionRef.current.kill();
      transitionRef.current = null;
    }
    setIsTransitioning(false);
  }, []);

  return { isTransitioning, navigateTo, cancelTransition };
}
```

```tsx
"use client";

import { useEffect, useState, useRef } from "react";
import { HeroBackground, HeroBackgroundHandle } from "../components/HeroBackground";
import { HeroMenu } from "../components/HeroMenu";
import { HeroTransitionOverlay } from "../components/HeroTransitionOverlay";
import { useReducedMotion } from "../providers/ReducedMotionProvider";
import { useHeroTransition } from "../hooks/useHeroTransition";
import { VIDEO_TARGETS } from "../lib/constants";

export function HeroSection() {
  const { isReducedMotion } = useReducedMotion();
  const heroBgRef = useRef<HeroBackgroundHandle>(null);
  const videoRef = useRef<HTMLVideoElement | null>(null);
  const [videoDuration, setVideoDuration] = useState(0);

  // Récupérer la durée vidéo et la ref vidéo
  useEffect(() => {
    const handle = heroBgRef.current;
    if (!handle) return;

    const interval = setInterval(() => {
      const video = handle.getVideoElement();
      const duration = handle.getDuration();
      if (video && duration > 0) {
        videoRef.current = video;
        setVideoDuration(duration);
        clearInterval(interval);
      }
    }, 100);

    return () => clearInterval(interval);
  }, []);

  const { isTransitioning, navigateTo } = useHeroTransition({
    videoRef,
    videoDuration,
    isReducedMotion,
  });

  const handleNavigate = (sectionId: string) => {
    const targetProgress = VIDEO_TARGETS[sectionId] ?? 0.05;

    if (sectionId === "continue") {
      // "Continuer" = débloquer le scroll (comportement actuel)
      // On scrub légèrement la vidéo puis on laisse le scroll natif reprendre
      navigateTo(targetProgress);
      return;
    }

    // Pour les autres : scrub vidéo + scroll vers section
    navigateTo(targetProgress, sectionId);
  };

  return (
    <section id="hero" className="relative min-h-[200vh] overflow-hidden">
      <div className="sticky top-0 h-screen w-full">
        {/* Arrière-plan Hero avec vidéo */}
        <HeroBackground ref={heroBgRef} />

        {/* Overlay de transition */}
        <HeroTransitionOverlay isVisible={isTransitioning} />

        {/* Contenu du menu CP2077 */}
        <div className="absolute inset-0 z-10 flex flex-col items-center justify-end pb-8 md:items-start md:justify-center md:pb-0">
          <HeroMenu onNavigate={handleNavigate} isTransitioning={isTransitioning} />
        </div>
      </div>
    </section>
  );
}
```
